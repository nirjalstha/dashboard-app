Google Sheets Bi-Directional Integration Setup

Follow these 3 easy steps to connect this frontend dashboard directly to your own Google Sheet using Google Apps Script (100% Free & No Backend Server Required).

1 Create Google Sheet Columns

Create a new Google Sheet and add these exact header columns in Row 1:

ID | Date | Title | Category | Region | Amount | Status
2 Add Apps Script Webhook Code

In your Google Sheet, click Extensions > Apps Script. Replace all code with this script:

function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var rows = sheet.getDataRange().getValues();
  var headers = rows[0];
  var data = [];
  
  for (var i = 1; i < rows.length; i++) {
    var row = rows[i];
    if (!row[0]) continue;
    var record = {};
    for (var j = 0; j < headers.length; j++) {
      record[headers[j].toString().toLowerCase()] = row[j];
    }
    data.push(record);
  }
  
  return ContentService.createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var payload = JSON.parse(e.postData.contents);
  
  // Append row
  sheet.appendRow([
    payload.id || "REC-" + Date.now().toString().slice(-4),
    payload.date,
    payload.title,
    payload.category,
    payload.region,
    parseFloat(payload.amount),
    payload.status
  ]);
  
  return ContentService.createTextOutput(JSON.stringify({ status: "success" }))
    .setMimeType(ContentService.MimeType.JSON);
}
3 Deploy as Web App
  1. Click Deploy > New deployment in Apps Script.
  2. Select type: Web App.
  3. Set Who has access to: Anyone.
  4. Click Deploy and copy the Web App URL.
  5. Paste URL into this app's Configuration Panel ().
